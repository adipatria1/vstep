// ══════════════════════════════════════════════════════════
// 1. VOLATILITY ENGINE
// ══════════════════════════════════════════════════════════
stdevVal      = ta.stdev(close, volatilityLength)
atrVal        = ta.atr(volatilityLength)
volValue      = (volatilityType == 'Stdev' ? stdevVal : atrVal) * volMult
channelOffset = volValue * channelMult

// ══════════════════════════════════════════════════════════
// 2. STEP S/R LEVEL
// ══════════════════════════════════════════════════════════
var float confirmedSr = close

float stepLevel = lineMode == 'Anchor' ? close : confirmedSr + (close > confirmedSr ? 1 : -1) * volValue
float newLevel  = math.abs(close - confirmedSr) > volValue ? stepLevel : confirmedSr

if barstate.isconfirmed
    confirmedSr := newLevel

float activeSr = realtimeUpdate ? newLevel : confirmedSr

// ══════════════════════════════════════════════════════════
// 3. HORIZONTAL LINE & LABEL RENDERING
// ══════════════════════════════════════════════════════════
newLine(y, col, w, sty) =>
    line.new(bar_index, y, bar_index + 1, y, color = col, width = w, style = sty, extend = extendActive ? extend.right : extend.none)

closeLine(lineId) =>
    if not na(lineId)
        if extendActive
            line.set_extend(lineId, extend.none)
        line.set_x2(lineId, bar_index)

extendLine(lineId) =>
    if not na(lineId) and not extendActive
        line.set_x2(lineId, bar_index + 1)

signedPct(v) =>
    (v > 0 ? '+' : '') + str.tostring(v, pctFormat) + '%'

formatLabelText(curr, prev, pct, hi, lo) =>
    string priceStr = (showSrPrefix ? 'SR ' : '') + str.tostring(curr, format.mintick)
    string base     = ''

    if na(prev) or na(pct)
        base := priceStr
    else
        pctStr   = signedPct(pct)
        diff     = curr - prev
        diffSign = diff > 0 ? '+' : ''
        diffStr  = diffSign + str.tostring(diff, format.mintick)

        base := switch labelContent
            'Price Only'             => priceStr
            'Price & Percent'        => priceStr + ' (' + pctStr + ')'
            'Percent Only'           => pctStr
            'Diff & Percent'         => diffStr + ' (' + pctStr + ')'
            'Full (Price, Diff & %)' => priceStr + ' [' + diffStr + '] (' + pctStr + ')'
            => priceStr + ' (' + pctStr + ')'

    string result = base
    if showRange and not na(hi) and not na(lo) and curr != 0
        float hiPct  = (hi - curr) / curr * 100
        float loPct  = (lo - curr) / curr * 100
        string hiStr = 'H ' + signedPct(hiPct)
        string loStr = 'L ' + signedPct(loPct)
        result := rangeLayout == 'Multi-line' ? base + '\n' + hiStr + '\n' + loStr : base + '  |  ' + hiStr + '  ' + loStr
    result

var line  midLine   = na
var line  upLine    = na
var line  loLine    = na
var label midLabel  = na
var float curLevel  = na
var float prevLevel = na
var float curPct    = na
var float srHigh    = na
var float srLow     = na

levelChanged = na(curLevel) or activeSr != curLevel or (showChannel and na(upLine) and not na(channelOffset))

if levelChanged
    closeLine(midLine)
    closeLine(upLine)
    closeLine(loLine)

    if not na(midLabel)
        if labelMode == 'Active Only'
            label.delete(midLabel)
            midLabel := na
        else
            label.set_x(midLabel, bar_index)

    bool isPriceStep = na(curLevel) or activeSr != curLevel
    if isPriceStep
        if not na(curLevel)
            prevLevel := curLevel
        curLevel := activeSr
        srHigh   := na
        srLow    := na

    midLine := newLine(activeSr, lineColor, lineWidth, line.style_solid)

    if showChannel and not na(channelOffset)
        upLine := newLine(activeSr + channelOffset, color.new(lineColor, 20), 1, channelStyle)
        loLine := newLine(activeSr - channelOffset, color.new(lineColor, 20), 1, channelStyle)
    else
        upLine := na
        loLine := na

    if showLabels
        float pctChange = not na(prevLevel) and prevLevel != 0 ? ((activeSr - prevLevel) / prevLevel) * 100 : na
        curPct := pctChange
        bool  isUp      = nz(pctChange) >= 0
        color boxCol    = lblStyle == label.style_none ? color.new(color.white, 100) : (colorMode == 'Directional (Green/Red)' ? (isUp ? bullColor : bearColor) : lineColor)
        color txtCol    = lblStyle == label.style_none ? (colorMode == 'Directional (Green/Red)' ? (isUp ? bullColor : bearColor) : lineColor) : color.white
        string txt      = formatLabelText(activeSr, prevLevel, pctChange, srHigh, srLow)

        midLabel := label.new(
            x         = bar_index + labelOffset,
            y         = activeSr,
            text      = txt,
            color     = boxCol,
            textcolor = txtCol,
            style     = lblStyle,
            size      = lblSize,
            textalign = text.align_left
        )
    else
        midLabel := na
else
    srHigh := na(srHigh) ? high : math.max(srHigh, high)
    srLow  := na(srLow)  ? low  : math.min(srLow, low)

    extendLine(midLine)
    extendLine(upLine)
    extendLine(loLine)
    if showLabels and not na(midLabel)
        label.set_x(midLabel, bar_index + labelOffset)
        label.set_text(midLabel, formatLabelText(curLevel, prevLevel, curPct, srHigh, srLow))

type Pos
    int    side          // 1 = Long, -1 = Short
    float  entry
    float  qty
    float  margin
    float  feeIn         // Fee entry (Taker/Market) yang sudah dibayar saat buka posisi
    float  tp            // na = tanpa TP harga
    float  sl            // na = tanpa SL harga tetap
    int    tpBarT        // 0 = bukan Bar TP | 3 = BarFix | 4 = BarProf | 5 = BarDir
    int    tpBarN        // target N bar / candle searah untuk TP
    int    dirCnt        // hitungan candle searah berturut-turut (bar closed)
    string tpDesc        // label info TP pada ENTRY
    int    slBarT        // 0 = bukan Bar SL | 3 = BarLoss | 4 = MaxBar | 5 = BarAdv
    int    slBarN        // target N bar / candle lawan untuk SL
    int    advCnt        // hitungan candle melawan arah berturut-turut (bar closed)
    string slDesc        // label info SL pada ENTRY
    float  actPct        // na = tanpa trailing
    float  distPct
    float  ext           // harga ekstrem sejak entry
    float  tstop         // trailing stop aktif (na = belum aktif)
    int    bar           // bar_index saat entry
    int    entryTime     // timestamp bar SR saat entry
    int    lvlStartTime  // timestamp bar ke-2 (mulai evaluasi TP/SL)
    float  trTrigPx      // harga aktivasi trailing
    int    trActTime     // timestamp saat trailing mulai aktif
    float  trActStop     // nilai tstop pertama saat trailing mulai aktif

type TradeRec

    float  movePct       // % pergerakan harga murni
    float  netPnl        // Net PnL bersih (USDT) setelah dikurangi Fee Entry + Fee Exit

type Sim

// Helper klasifikasi kategori strategi untuk Filter Leaderboard
method isBarTp(Sim s) =>
    s.tpT >= 3

method isBarSl(Sim s) =>
    s.slT >= 3

method isBarAny(Sim s) =>
    s.tpT >= 3 or s.slT >= 3

method isBarBoth(Sim s) =>
    s.tpT >= 3 and s.slT >= 3

method isFixedPct(Sim s) =>
    s.tpT < 3 and s.slT < 3 and (s.tpT == 2 or s.slT == 2 or (s.useTr and s.trU == 2))

method isMedianPct(Sim s) =>
    s.tpT < 3 and s.slT < 3 and (s.tpT == 1 or s.slT == 1 or (s.useTr and s.trU == 1))

method isPctAny(Sim s) =>
    s.tpT < 3 and s.slT < 3 and (s.tpT > 0 or s.slT > 0 or s.useTr)

scanAt(Sim s, float px) =>
    float f = 0.0
    float n = 0.0
    for p in s.book
        f += p.side * p.qty * (px - p.entry)
        n += p.qty * px
    [f, n]

method usedM(Sim s) =>
    float t = 0.0
    for p in s.book
        t += p.margin
    t

method closeAt(Sim s, int i, float px, float feePct, int reason, string why) =>
    Pos   p        = s.book.get(i)
    float grossPnl = p.side * p.qty * (px - p.entry)
    float feeOut   = p.qty * px * feePct / 100
    // Saldo ditambah grossPnl dikurangi feeOut (karena p.feeIn sudah dipotong dari saldo saat entry)
    s.bal  += grossPnl - feeOut
    s.fees += feeOut
    // Net PnL bulat per trade dikurangi total fee 2 arah (Entry + Exit)
    float net = grossPnl - p.feeIn - feeOut
    if net > 0
        s.nWin += 1
        s.gp   += net
    else
        s.nLoss += 1
        s.gl    += -net
    if reason == 1
        s.nTp += 1
    else if reason == 5
        s.nTp    += 1
        s.nBarTp += 1
    else if reason == 2
        s.nStop += 1
    else if reason == 6
        s.nStop  += 1
        s.nBarSl += 1
    else if reason == 3
        s.nStop  += 1
        s.nTrail += 1
    else
        s.nNext += 1

    float movePct = p.side * (px - p.entry) / p.entry * 100
    int   lvlSt   = na(p.lvlStartTime) ? time : p.lvlStartTime
    s.trades.push(TradeRec.new(
        p.side, p.bar, p.entryTime, lvlSt, p.entry, p.tp, p.sl, p.tpDesc, p.slDesc,
        not na(p.actPct), p.trTrigPx, p.trActTime, p.trActStop, p.tstop,
        bar_index, time, px, reason, why, movePct, net
    ))
    if s.trades.size() > maxVisTrades
        s.trades.shift()

    s.book.remove(i)
    true

method processBar(Sim s, bool newSr, float srPx, float prevSr, float medH, float medL, int nSamp, bool inWin) =>
    if not s.liquid
        float minEqPre = s.bal

        // 1. Evaluasi posisi yang sudah terbuka dari bar sebelumnya
        if s.book.size() > 0
            [fLo, nLo] = scanAt(s, low)
            [fHi, nHi] = scanAt(s, high)
            float eqLo = s.bal + fLo
            float eqHi = s.bal + fHi
            minEqPre := math.min(eqLo, eqHi)

            if eqLo <= nLo * mmrPct / 100 or eqHi <= nHi * mmrPct / 100
                s.liquid  := true
                s.liqTime := time
                s.liqBar  := bar_index
                s.liqPx   := close
                s.bal     := 0.0
                s.ddEq    := 100.0
                s.ddBal   := 100.0
                s.book.clear()
            else
                int i = s.book.size() - 1
                while i >= 0
                    Pos p = s.book.get(i)
                    if na(p.lvlStartTime)
                        p.lvlStartTime := time

                    int barsHeld = bar_index - p.bar

                    // Hitung candle searah & melawan arah hanya saat candle resmi tutup (barstate.isconfirmed)
                    if barstate.isconfirmed
                        bool isDirCandle = p.side == 1 ? close > open : close < open
                        bool isAdvCandle = p.side == 1 ? close < open : close > open
                        p.dirCnt := isDirCandle ? p.dirCnt + 1 : 0
                        p.advCnt := isAdvCandle ? p.advCnt + 1 : 0

                    float effStop = na
                    if p.side == 1
                        effStop := na(p.sl) ? p.tstop : (na(p.tstop) ? p.sl : math.max(p.sl, p.tstop))
                    else
                        effStop := na(p.sl) ? p.tstop : (na(p.tstop) ? p.sl : math.min(p.sl, p.tstop))
                    bool stopHit = not na(effStop) and (p.side == 1 ? low <= effStop : high >= effStop)
                    bool tpHit   = not na(p.tp) and (p.side == 1 ? high >= p.tp : low <= p.tp)

                    // Evaluasi Bar-Based TP & Bar-Based SL murni saat candle resmi tutup (barstate.isconfirmed)
                    bool   barTpHit = false
                    string barTpWhy = ''
                    bool   barSlHit = false
                    string barSlWhy = ''

                    if barstate.isconfirmed
                        float closeGross  = p.side * p.qty * (close - p.entry)
                        float closeFeeOut = p.qty * close * feeEntry / 100
                        float closeNetPnl = closeGross - p.feeIn - closeFeeOut
                        bool  isNetProf   = closeNetPnl > 0.0
                        bool  isInLoss    = closeNetPnl <= 0.0

                        if p.tpBarT == 3 and barsHeld >= p.tpBarN
                            barTpHit := true
                            barTpWhy := 'TP BAR (' + str.tostring(barsHeld) + 'b)'
                        else if p.tpBarT == 4 and barsHeld >= p.tpBarN and isNetProf
                            barTpHit := true
                            barTpWhy := 'TP B-PROF (' + str.tostring(barsHeld) + 'b)'
                        else if p.tpBarT == 5 and p.dirCnt >= p.tpBarN and isNetProf
                            barTpHit := true
                            barTpWhy := 'TP B-DIR (' + str.tostring(p.dirCnt) + 'c)'

                        if p.slBarT == 3 and barsHeld >= p.slBarN and isInLoss
                            barSlHit := true
                            barSlWhy := 'SL B-LOSS (' + str.tostring(barsHeld) + 'b)'
                        else if p.slBarT == 4 and barsHeld >= p.slBarN
                            barSlHit := true
                            barSlWhy := 'SL MAX-BAR (' + str.tostring(barsHeld) + 'b)'
                        else if p.slBarT == 5 and p.advCnt >= p.slBarN and isInLoss
                            barSlHit := true
                            barSlWhy := 'SL B-ADV (' + str.tostring(p.advCnt) + 'c)'

                    // Urutan konservatif: SL Harga/Trail -> Bar SL (Loss/Adv) -> TP Harga -> Bar TP -> MaxBar Timeout
                    if stopHit
                        bool isTr = not na(p.tstop) and (na(p.sl) or (p.side == 1 ? p.tstop >= p.sl : p.tstop <= p.sl))
                        s.closeAt(i, effStop, feeEntry, isTr ? 3 : 2, isTr ? 'TRAIL' : 'SL')
                    else if barSlHit and p.slBarT != 4
                        s.closeAt(i, close, feeEntry, 6, barSlWhy)
                    else if tpHit
                        s.closeAt(i, p.tp, feeExit, 1, 'TP')
                    else if barTpHit
                        s.closeAt(i, close, feeEntry, 5, barTpWhy)
                    else if barSlHit
                        s.closeAt(i, close, feeEntry, 6, barSlWhy)
                    else if not na(p.actPct)
                        p.ext := p.side == 1 ? math.max(p.ext, high) : math.min(p.ext, low)
                        float favPct = p.side == 1 ? (p.ext - p.entry) / p.entry * 100 : (p.entry - p.ext) / p.entry * 100
                        if favPct >= p.actPct
                            float ns = p.side == 1 ? p.ext * (1 - p.distPct / 100) : p.ext * (1 + p.distPct / 100)
                            if na(p.tstop)
                                p.tstop     := ns
                                p.trActTime := time
                                p.trActStop := ns
                            else
                                p.tstop := p.side == 1 ? math.max(p.tstop, ns) : math.min(p.tstop, ns)
                    i -= 1

        if not s.liquid
            // 2. Penanganan Posisi Lama Saat Terbentuk SR Baru
            if newSr and s.book.size() > 0
                if s.slT == 0
                    // Strategi SL NextSR: wajib tutup semua posisi lama di harga SR baru
                    int j = s.book.size() - 1
                    while j >= 0
                        s.closeAt(j, srPx, feeEntry, 4, 'NEXT SR')
                        j -= 1
                else if s.mode != 0 and not na(prevSr)
                    // Strategi One-Way Mode (Trend/Counter) dengan SL Tetap / Bar-SL: pastikan maksimal 1 posisi!
                    bool upStepNow = srPx > prevSr
                    int  newSigDir = s.mode == 1 ? (upStepNow ? 1 : -1) : (upStepNow ? -1 : 1)
                    if oneWayRule == 'Tutup Posisi Lama & Buka Baru di SR Baru'
                        int j = s.book.size() - 1
                        while j >= 0
                            s.closeAt(j, srPx, feeEntry, 4, 'NEXT SR')
                            j -= 1
                    else if oneWayRule == 'Reverse Jika Lawan Arah, Tahan Jika Searah'
                        int j = s.book.size() - 1
                        while j >= 0
                            if s.book.get(j).side != newSigDir
                                s.closeAt(j, srPx, feeEntry, 4, 'REV SR')
                            j -= 1

            // 3. Entry Baru Tepat di Bar & Harga SR Saat Terbentuk
            if newSr and not na(prevSr) and inWin
                bool needMed       = s.tpT == 1 or s.slT == 1 or (s.useTr and s.trU == 1)
                int  maxPosAllowed = s.mode == 0 ? maxOpen : 1
                if not needMed or nSamp >= minLevels
                    for k = 0 to 1
                        int  dir     = k == 0 ? 1 : -1
                        bool upStep  = srPx > prevSr
                        bool allowed = s.mode == 0 ? true : (s.mode == 1 ? (upStep ? dir == 1 : dir == -1) : (upStep ? dir == -1 : dir == 1))
                        if allowed
                            float favM    = dir == 1 ? medH : medL
                            float advM    = dir == 1 ? medL : medH
                            bool  isBarTp = s.tpT >= 3
                            bool  isBarSl = s.slT >= 3
                            float tpP     = s.tpT == 1 ? favM * s.tpV : (s.tpT == 2 ? s.tpV : na)
                            float slP     = s.slT == 1 ? advM * s.slV : (s.slT == 2 ? s.slV : na)
                            float acP     = s.useTr ? (s.trU == 1 ? favM * s.trA : s.trA) : na
                            float dsP     = s.useTr ? (s.trU == 1 ? favM * s.trD : s.trD) : na
                            bool  ok      = (s.tpT == 0 or isBarTp or tpP > 0) and (s.slT == 0 or isBarSl or slP > 0) and (not s.useTr or (acP > 0 and dsP > 0))

                            float base     = marginBasis == 'Initial Balance' ? initCap : s.bal
                            float mrg      = base * marginPct / 100
                            float notional = mrg * leverage
                            float feeIn    = notional * feeEntry / 100
                            [fC, nC]       = scanAt(s, close)
                            float availEq  = s.bal + fC - s.usedM()

                            if ok
                                if mrg > 0 and mrg + feeIn <= availEq and s.book.size() < maxPosAllowed
                                    float  qty      = notional / srPx
                                    float  tpPx     = na(tpP) ? na : srPx * (1 + dir * tpP / 100)
                                    float  slPx     = na(slP) ? na : srPx * (1 - dir * slP / 100)
                                    float  trTrigPx = na(acP) ? na : srPx * (1 + dir * acP / 100)
                                    int    bTpType  = isBarTp ? s.tpT : 0
                                    int    bTpTgt   = isBarTp ? int(s.tpV) : 0
                                    int    bSlType  = isBarSl ? s.slT : 0
                                    int    bSlTgt   = isBarSl ? int(s.slV) : 0
                                    string tpLabel  = not na(tpPx) ? str.tostring(tpPx, format.mintick) : (s.useTr ? 'TRAIL' : (bTpType == 3 ? 'BarFix(' + str.tostring(bTpTgt) + 'b)' : (bTpType == 4 ? 'BarProf(>=' + str.tostring(bTpTgt) + 'b)' : (bTpType == 5 ? 'BarDir(' + str.tostring(bTpTgt) + 'c)' : '-'))))
                                    string slLabel  = not na(slPx) ? str.tostring(slPx, format.mintick) : (bSlType == 3 ? 'BarLoss(>=' + str.tostring(bSlTgt) + 'b)' : (bSlType == 4 ? 'MaxBar(' + str.tostring(bSlTgt) + 'b)' : (bSlType == 5 ? 'BarAdv(' + str.tostring(bSlTgt) + 'c)' : 'NextSR')))
                                    s.book.push(Pos.new(dir, srPx, qty, mrg, feeIn, tpPx, slPx, bTpType, bTpTgt, 0, tpLabel, bSlType, bSlTgt, 0, slLabel, acP, dsP, srPx, float(na), bar_index, time, int(na), trTrigPx, int(na), float(na)))
                                    s.bal  -= feeIn
                                    s.fees += feeIn
                                    if dir == 1
                                        s.nLong += 1
                                    else
                                        s.nShort += 1
                                else
                                    s.nSkip += 1

            // 4. Max Drawdown (equity worst-case & balance)
            [fCl, nCl] = scanAt(s, close)
            float eqClose = s.bal + fCl
            float worst   = math.min(minEqPre, eqClose)
            s.ddEq    := math.max(s.ddEq, s.peakEq > 0 ? (s.peakEq - worst) / s.peakEq * 100 : 0.0)
            s.peakEq  := math.max(s.peakEq, eqClose)
            s.ddBal   := math.max(s.ddBal, s.peakBal > 0 ? (s.peakBal - s.bal) / s.peakBal * 100 : 0.0)
            s.peakBal := math.max(s.peakBal, s.bal)
    true

method netPct(Sim s) =>
    [f, n] = scanAt(s, close)
    ((s.liquid ? 0.0 : s.bal + f) - initCap) / initCap * 100

method pf(Sim s) =>
    s.gl > 0 ? s.gp / s.gl : (s.gp > 0 ? 99.0 : 0.0)

method score(Sim s) =>
    int   n  = s.nWin + s.nLoss
    float nt = s.netPct()
    n == 0 ? -1000000000.0 : (rankBy == 'Net Profit %' ? nt : (rankBy == 'Net / Max DD' ? nt / math.max(s.ddEq, 1.0) : s.pf()))

mk(string nm, string key, int mode, int tpT, float tpV, int slT, float slV, bool useTr, int trU, float trA, float trD) =>
    Sim s = Sim.new(name = nm, mode = mode, tpT = tpT, tpV = tpV, slT = slT, slV = slV, useTr = useTr, trU = trU, trA = trA, trD = trD, key = key)
    s.book    := array.new<Pos>()
    s.trades  := array.new<TradeRec>()
    s.bal     := initCap
    s.peakEq  := initCap
    s.peakBal := initCap
    s
